Image Compression Function

The function compress_image takes in an image in the format of a \(n \times p\) matrix and a value \(k\) for the \(k^{th}\) rank approximation and returns:

  1. The rank \(k\) approximation to the image using PCA
  2. the error of rank \(k\) approximation according to the Frobenius norm
Code 
compress_image_from_string <- function(img_name, k){
  img <- read_csv(img_name)
  mat <- data.matrix(img)
  
  return(compress_image_from_mat(mat, k))
}
# same method above excpet image is already a matrix 
compress_image_from_mat <- function(mat, k){
  # cross product of mat (X'X) so that we can do 
  # V = UΛU'
  V <- t(mat) %*% mat
  # eigendecomposition 
  decomp <- eigen(V)
  U <- decomp$vectors
  Z <- mat %*% U
  # rank approximation
  rank_approx <- matrix(0, nrow = nrow(mat), ncol = ncol(mat))
  for (i in 1:k) {
    rank_approx <- rank_approx + Z[, i] %*% t(U[, i])
  }
  # error using Frobenius norm
  error <- norm(mat - rank_approx, type="F")
  # save the results as a list
  # to return multiple objects
  results <- list("approx" = rank_approx, "error" = error)
  
  return(results)
}

show_image_csv <- function(img_name){
  img <- read_csv(img_name)
  mat <- data.matrix(img)
  
  return(show_image_mat(mat))
}

show_image_mat <- function(img_mat) {

  img_plot <- t(img_mat[nrow(img_mat):1, ])

  df <- expand.grid(
    x = seq_len(nrow(img_plot)),
    y = seq_len(ncol(img_plot))
  )

  df$value <- as.vector(img_plot)

  ggplot(df, aes(x = x, y = y, fill = value)) +
    geom_raster() +
    scale_fill_gradientn(
      colours = gray.colors(256)
    ) +
    scale_x_continuous(expand = c(0, 0)) +
    scale_y_continuous(expand = c(0, 0)) +
    coord_fixed() + 
    theme_void() +
    theme(
      legend.position = "none"
    )
}

process_img <- function(img_name){
  # plot the original image
  img_mat <- data.matrix(read_csv(img_name))
  
  original_img <- show_image_mat(img_mat)
  
  # plot of error vs k for all possible k values
  p <- ncol(img_mat)
  error_list <- numeric(p)
  
  approximation_plots <- list()
  for (i in 1:p){
    results <- compress_image_from_mat(img_mat, i)
    
    # plot first 10 pca apprimixations
    if (i <= 10){
      approx <- results$approx
      plot <- show_image_mat(approx) + 
        labs(title = sprintf("k = %d Approximation", i)) +
        theme(
          plot.title = element_text(
            hjust = 0.5,
            size = 18
          )
        )
      
      approximation_plots[[i]] <- plot
    }
    
    error_list[i] <- results$error
  }
  
  df <- data.frame(
    "error" = error_list, 
    "k" = 1:p
  )
  
  plot_title <- sprintf("Error vs Kth Approximation (%s)", img_name)
  
  error_plot <- df |> ggplot(aes(x=k, y=error)) + 
    geom_point(size=1) + 
    scale_x_continuous(
      name="Kth Approximation",
      breaks = seq(0, p, 15)
    ) + 
    labs(
      title=plot_title, 
      y="Error"
    )
  
  # plot of the first eigenvector and the first pc score
  V <- t(img_mat) %*% img_mat
  decomp <- eigen(V)
  U <- decomp$vectors
  Z <- img_mat %*% U
  
  first_eigenvector <- U[, 1]
  first_pc_score <- Z[, 1]
  
  df_eigenvector <- data.frame(
    column_index = 1:length(first_eigenvector), 
    value = first_eigenvector
  )
  df_pc_score <- data.frame(
    row_index = 1:length(first_pc_score), 
    value = first_pc_score
  )
  
  p1 <- df_eigenvector |> ggplot(aes(x=column_index, y=value)) + 
    geom_line(color="blue", size=0.8) + 
    labs(
      title = "First Eigenvector", 
      x="Column Index (Image Width)", 
      y="Weight Value"
    ) + 
    theme_minimal()
  
  p2 <- df_pc_score |> ggplot(aes(x=row_index, y=value)) + 
    geom_line(color ="darkred", size=0.8) + 
    scale_x_reverse() +
    coord_flip() + 
    labs(
      title="First PC Score", 
      x="Row Index (Image Height)", 
      y="Score Value"
    ) + 
    theme_minimal()
  
  first_eig_pc <- (p1 + p2)
  
  list(
    original = original_img,
    approximations = approximation_plots,
    error = error_plot,
    first_eig_pc = first_eig_pc
  )
  
}

Interpretation

The first eigenvector graph is a smooth line with a negative slope. This suggests that the first eigenvector captures the main horizontal gradient in the image, as the image changes from one brightness level on the left to another on the right. The larger magnitude of the weights toward the right side shows that this part of the image contributes strongly to this horizontal pattern.

PC Score 1 tracks how this horizontal pattern changes vertically. Because the first eigenvector gives more weight to the right side of the image, the large spikes in the PC score correspond to rows containing the sharp horizontal lines on the right side.

Best Value of K

I think k=1 is sufficiently good because the rank-1 approximation already preserves the main gradient and horizontal structures visible in the original image. Higher values of k add relatively minor details, so k=1 provides a strong amount of compression while retaining the important features.
